cmake_minimum_required(VERSION 3.20)
project(ninacal_backend LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(ZLIB REQUIRED)

add_executable(ninacal-server src/main.cpp src/store.cpp src/region.cpp)
target_include_directories(ninacal-server PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_compile_definitions(ninacal-server PRIVATE CPPHTTPLIB_ZLIB_SUPPORT)
target_compile_options(ninacal-server PRIVATE -Wall -Wextra -Wpedantic)
if(TARGET SQLite3::SQLite3)
  set(NINACAL_SQLITE SQLite3::SQLite3)
else()
  set(NINACAL_SQLITE SQLite::SQLite3)
endif()
target_link_libraries(ninacal-server PRIVATE Threads::Threads ${NINACAL_SQLITE} ZLIB::ZLIB)
